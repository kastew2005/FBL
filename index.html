// Обновленный TypeScript-код с гравитацией и физикой обрушения блоков (Teardown style) // Блоки теряют опору, падают вниз под действием гравитации и разрушаются при падении. document.body.style.margin = '0'; document.body.style.overflow = 'hidden'; document.body.style.background = '#111'; const canvas = document.createElement('canvas'); canvas.width = window.innerWidth; canvas.height = window.innerHeight; document.body.appendChild(canvas); const ctx = canvas.getContext('2d')!; interface Block { x: number; y: number; color: string; health: number; maxHealth: number; } interface FallingBlock { x: number; y: number; vx: number; vy: number; color: string; size: number; } interface Particle { x: number; y: number; vx: number; vy: number; color: string; life: number; } class TeardownGravityGame { private player = { x: canvas.width / 2, y: canvas.height / 2, size: 16, speed: 3 }; private blocks: (Block | null)[][] = []; private fallingBlocks: FallingBlock[] = []; private particles: Particle[] = []; private keys: { [key: string]: boolean } = {}; private touchMove = { x: 0, y: 0, active: false }; private readonly GRID_SIZE = 24; private cols = Math.ceil(canvas.width / this.GRID_SIZE); private rows = Math.ceil(canvas.height / this.GRID_SIZE); constructor() { this.initMap(); this.initInput(); this.loop(); } private initMap() { this.blocks = []; const colors = ['#555', '#777', '#8B4513', '#A0522D', '#CD853F']; for (let r = 0; r < this.rows; r++) { this.blocks[r] = []; for (let c = 0; c < this.cols; c++) { const isWall = (r === 0 || r === this.rows - 1 || c === 0 || c === this.cols - 1 || (Math.random() < 0.25 && (Math.abs(c - this.cols/2) > 3 || Math.abs(r - this.rows/2) > 3))); if (isWall) { const maxHp = 3; this.blocks[r][c] = { x: c * this.GRID_SIZE, y: r * this.GRID_SIZE, color: colors[Math.floor(Math.random() * colors.length)], health: maxHp, maxHealth: maxHp }; } else { this.blocks[r][c] = null; } } } } private initInput() { window.addEventListener('keydown', e => this.keys[e.code] = true); window.addEventListener('keyup', e => this.keys[e.code] = false); canvas.addEventListener('touchstart', e => this.handleTouch(e), { passive: false }); canvas.addEventListener('touchmove', e => this.handleTouch(e), { passive: false }); canvas.addEventListener('touchend', e => this.handleTouch(e), { passive: false }); canvas.addEventListener('mousedown', e => this.destroyAt(e.clientX, e.clientY)); } private handleTouch(e: TouchEvent) { e.preventDefault(); this.touchMove.active = false; for (let i = 0; i < e.touches.length; i++) { const t = e.touches[i]; if (t.clientX < canvas.width / 2) { this.touchMove = { x: t.clientX, y: t.clientY, active: true }; } else { this.destroyAt(t.clientX, t.clientY); } } } private destroyAt(px: number, py: number) { const c = Math.floor(px / this.GRID_SIZE); const r = Math.floor(py / this.GRID_SIZE); if (r >= 0 && r < this.rows && c >= 0 && c < this.cols) { const block = this.blocks[r][c]; if (block) { block.health -= 3; // Уничтожаем блок кувалдой this.spawnParticles(px, py, block.color, 6); if (block.health <= 0) { this.blocks[r][c] = null; this.checkGravityCollapse(r, c); // Запускаем проверку обрушения по вертикали } } } } // Проверка гравитации: если под блоком пусто, он превращается в падающий объект private checkGravityCollapse(destroyedR: number, destroyedC: number) { // Проверяем все блоки, находящиеся строго выше разрушенного в том же столбце for (let r = destroyedR - 1; r >= 0; r--) { const block = this.blocks[r][destroyedC]; if (block) { // Превращаем висящий блок в падающий под действием гравитации this.fallingBlocks.push({ x: block.x, y: block.y, vx: (Math.random() - 0.5) * 1.5, vy: 0, color: block.color, size: this.GRID_SIZE - 2 }); this.blocks[r][destroyedC] = null; } else { break; // Если встретили пустоту, выше цепочка не падает } } } private spawnParticles(x: number, y: number, color: string, count: number) { for (let i = 0; i < count; i++) { this.particles.push({ x, y, vx: (Math.random() - 0.5) * 5, vy: (Math.random() - 0.5) * 5, color, life: 25 }); } } private update() { let dx = 0, dy = 0; if (this.keys['KeyW'] || this.keys['ArrowUp']) dy -= 1; if (this.keys['KeyS'] || this.keys['ArrowDown']) dy += 1; if (this.keys['KeyA'] || this.keys['ArrowLeft']) dx -= 1; if (this.keys['KeyD'] || this.keys['ArrowRight']) dx += 1; if (this.touchMove.active) { dx = (this.touchMove.x - canvas.width / 4) / 50; dy = (this.touchMove.y - canvas.height / 2) / 50; const len = Math.hypot(dx, dy); if (len > 1) { dx /= len; dy /= len; } } // Движение игрока с коллизией const nextX = this.player.x + dx * this.player.speed; const nextY = this.player.y + dy * this.player.speed; const colIdx = Math.floor(nextX / this.GRID_SIZE); const rowIdx = Math.floor(nextY / this.GRID_SIZE); if (rowIdx >= 0 && rowIdx < this.rows && colIdx >= 0 && colIdx < this.cols && !this.blocks[rowIdx][colIdx]) { this.player.x = nextX; this.player.y = nextY; } // Физика падающих блоков (Гравитация) for (let i = this.fallingBlocks.length - 1; i >= 0; i--) { const fb = this.fallingBlocks[i]; fb.vy += 0.4; // Ускорение свободного падения fb.x += fb.vx; fb.y += fb.vy; // Столкновение с полом или нижними границами сетки const targetR = Math.floor((fb.y + fb.size) / this.GRID_SIZE); const targetC = Math.floor((fb.x + fb.size / 2) / this.GRID_SIZE); if (targetR >= this.rows - 1 || (targetR >= 0 && targetR < this.rows && targetC >= 0 && targetC < this.cols && this.blocks[targetR][targetC])) { // Приземление: превращаем обратно в статический блок или разбиваем в пыль if (targetR >= 0 && targetR < this.rows && targetC >= 0 && targetC < this.cols) { const safeR = Math.min(targetR, this.rows - 2); if (!this.blocks[safeR][targetC]) { this.blocks[safeR][targetC] = { x: targetC * this.GRID_SIZE, y: safeR * this.GRID_SIZE, color: fb.color, health: 2, maxHealth: 3 }; } } this.spawnParticles(fb.x, fb.y, fb.color, 4); this.fallingBlocks.splice(i, 1); } } // Обновление частиц for (let i = this.particles.length - 1; i >= 0; i--) { const p = this.particles[i]; p.x += p.vx; p.y += p.vy; p.life--; if (p.life <= 0) this.particles.splice(i, 1); } } private render() { ctx.fillStyle = '#151515'; ctx.fillRect(0, 0, canvas.width, canvas.height); // Отрисовка сетки блоков for (let r = 0; r < this.rows; r++) { for (let c = 0; c < this.cols; c++) { const b = this.blocks[r][c]; if (b) { ctx.fillStyle = b.color; ctx.fillRect(b.x, b.y, this.GRID_SIZE - 1, this.GRID_SIZE - 1); } } } // Отрисовка летящих/падающих под гравитацией блоков for (const fb of this.fallingBlocks) { ctx.fillStyle = fb.color; ctx.fillRect(fb.x, fb.y, fb.size, fb.size); } // Отрисовка частиц for (const p of this.particles) { ctx.fillStyle = p.color; ctx.fillRect(p.x, p.y, 2, 2); } // Игрок ctx.fillStyle = '#ffcc00'; ctx.beginPath(); ctx.arc(this.player.x, this.player.y, this.player.size / 2, 0, Math.PI * 2); ctx.fill(); // Подсказка ctx.fillStyle = 'rgba(255, 255, 255, 0.5)'; ctx.font = '13px sans-serif'; ctx.fillText('Снесите нижнюю часть стены, чтобы активировать гравитацию и обрушение!', 20, 30); } private loop = () => { this.update(); this.render(); requestAnimationFrame(this.loop); } } new TeardownGravityGame();
